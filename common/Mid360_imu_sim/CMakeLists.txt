cmake_minimum_required(VERSION 3.8)
project(livox_laser_simulation)
find_package(ament_cmake REQUIRED)
find_package(gazebo REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
add_library(livox_laser_simulation SHARED src/livox_points_plugin.cpp src/livox_ode_multiray_shape.cpp)
target_compile_features(livox_laser_simulation PUBLIC cxx_std_17)
target_include_directories(livox_laser_simulation PRIVATE include ${GAZEBO_INCLUDE_DIRS})
target_link_directories(livox_laser_simulation PRIVATE ${GAZEBO_LIBRARY_DIRS})
ament_target_dependencies(livox_laser_simulation gazebo_ros rclcpp sensor_msgs livox_ros_driver2)
target_link_libraries(livox_laser_simulation ${GAZEBO_LIBRARIES})
install(TARGETS livox_laser_simulation LIBRARY DESTINATION lib)
install(DIRECTORY meshes scan_mode urdf DESTINATION share/${PROJECT_NAME})
install(PROGRAMS script/pointcloud2livox.py DESTINATION lib/${PROJECT_NAME})
ament_environment_hooks(env-hooks/livox_plugin_path.dsv.in)
ament_package()
